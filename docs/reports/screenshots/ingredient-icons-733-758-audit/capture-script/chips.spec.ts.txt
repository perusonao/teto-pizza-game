async function glyphs(page: Page) {
  return page.evaluate(() => [...document.querySelectorAll("svg.ingredient-glyph")].map((e) => {
    const r = e.getBoundingClientRect(); const cs = getComputedStyle(e); const font = parseFloat(cs.fontSize);
    const vis = [...e.classList].find((c) => c.startsWith("ingredient-glyph--"))!.replace("ingredient-glyph--", "");
    const inChip = !!e.closest(".ingredient-chip"); const inStage = !!e.closest(".pizza-stage");
    // rotation/scale of ancestors
    let m = new DOMMatrix(); for (let n: Element | null = e; n && n !== document.body; n = n.parentElement) { const t = getComputedStyle(n).transform; if (t && t !== "none") m = new DOMMatrix(t).multiply(m); }
    const rot = Math.round(Math.atan2(m.b, m.a) * 180 / Math.PI);
    return { vis, where: inChip ? "tray" : inStage ? "stage" : "other", x: +r.x.toFixed(1), y: +r.y.toFixed(1), w: +r.width.toFixed(1), h: +r.height.toFixed(1), font, em: +(parseFloat(cs.width) / font).toFixed(3), rot, cssW: parseFloat(cs.width), filter: getComputedStyle(e.closest(".ingredient-piece-visual__emoji") ?? e).filter };
  }).filter((g) => g.where !== "other" ));
}
async function shot(page: Page, name: string, project: string, meas: any, tag: string) {
  mkdirSync(OUT, { recursive: true });
  await page.screenshot({ path: `${OUT}/${name}-${project}.png` });
  const g = await glyphs(page);
  meas[tag] = g;
  const st = page.locator(".pizza-stage").first();
  if (await st.count()) await st.screenshot({ path: `${OUT}/${name}-pizza-${project}.png` }).catch(() => {});
}
async function place(page: Page, picks: [RegExp, number][], spot0: number, off = 0) {
  let s = spot0;
  for (const [name, n] of picks) { await (await chipOnTrayOrPin(page, name)).click();
    for (let i = 0; i < n; i++) { const [x, y] = SPOTS[s++ % SPOTS.length]; await tapDoughPercent(page, x + off, y + off); } }
  return s;
}
