import { createRequire } from "node:module";
import { createHash } from "node:crypto";
import { writeFileSync, mkdirSync } from "node:fs";
import { test, expect, type Page } from "@playwright/test";
import { startTargetlessFreeCook } from "../e2e/support/startFreeCook";
import { baseSave, boot, guardNetwork, pickChip, safetyPreflight, completeDoughStep, tapDoughPercent, enterBakePaused, landNeedleAndTakeOut, KEY } from "./helpers";

const sharp = createRequire(import.meta.url)("sharp");
mkdirSync("out/m", { recursive: true });
const R: Record<string, unknown> = {};
const log = (m: string) => { writeFileSync("out/progress.log", `${new Date().toISOString()} ${m}\n`, { flag: "a" }); };
const sha = (b: Buffer) => createHash("sha256").update(b).digest("hex");
const FREE_BAKE = (v: number) => ({ start: v - 1, end: v + 1 });

async function freshRound(context: import("@playwright/test").BrowserContext, old: Page): Promise<Page> {
  await old.close();
  const p = await context.newPage();
  await p.goto("icons/icon-16.png");
  await p.evaluate(([k, v]) => localStorage.setItem(k, v), [KEY, JSON.stringify(baseSave())] as const);
  await p.goto("./"); await p.waitForSelector(".app-frame");
  await startTargetlessFreeCook(p); await p.waitForSelector(".pizza-stage");
  return p;
}
async function prepareIkura(page: Page, opts: { roePre?: number; shisoPre?: number } = {}) {
  await completeDoughStep(page);
  await page.getByRole("button", { name: /次へ/ }).click(); // SAUCE (NO_SAUCE: nothing to paint)
  await page.getByRole("button", { name: /次へ/ }).click(); // CHEESE
  expect(await pickChip(page, /クリームチーズ/)).toBe(true);
  await tapDoughPercent(page, 38, 50); await tapDoughPercent(page, 62, 50);
  await page.getByRole("button", { name: /次へ/ }).click(); // TOPPING
  expect(await pickChip(page, /サーモン/)).toBe(true);
  await tapDoughPercent(page, 38, 32); await tapDoughPercent(page, 62, 32); await tapDoughPercent(page, 50, 70);
  if (opts.roePre) { expect(await pickChip(page, /いくら/)).toBe(true); for (let i = 0; i < opts.roePre; i += 1) await tapDoughPercent(page, 30 + i * 12, 55); }
  if (opts.shisoPre) { expect(await pickChip(page, /大葉/)).toBe(true); for (let i = 0; i < opts.shisoPre; i += 1) await tapDoughPercent(page, 70, 52 + i * 8); }
}
async function bake(page: Page, v: number) { await enterBakePaused(page); await landNeedleAndTakeOut(page, FREE_BAKE(v)); }
async function placePost(page: Page, roe: number, shiso: number) {
  const pts = [[36, 36], [64, 36], [50, 60], [30, 62], [70, 62]];
  let i = 0;
  if (roe) { expect(await pickChip(page, /いくら/)).toBe(true); for (let k = 0; k < roe; k += 1) { await tapDoughPercent(page, pts[i][0], pts[i][1]); i += 1; } }
  if (shiso) { expect(await pickChip(page, /大葉/)).toBe(true); for (let k = 0; k < shiso; k += 1) { await tapDoughPercent(page, pts[i][0], pts[i][1]); i += 1; } }
}
async function resultText(page: Page) { await page.waitForSelector(".result-panel", { timeout: 15000 }); return (await page.locator(".result-panel").innerText()).replace(/\s+/g, " "); }
async function isolatedPost(page: Page, name: string) {
  const info = await page.evaluate(() => {
    const out: any = {};
    const roes = [...document.querySelectorAll(".pizza-post-layer .pizza-topping--salmon-roe")];
    const shisos = [...document.querySelectorAll(".pizza-post-layer .pizza-topping--shiso")];
    const pre = [...document.querySelectorAll(".pizza-dough .pizza-topping")].map((e) => e.className);
    const dough = document.querySelector(".pizza-dough") as HTMLElement;
    const layer = document.querySelector(".pizza-post-layer") as HTMLElement | null;
    const rb = (e: Element) => { const r = e.getBoundingClientRect(); return { x: +r.x.toFixed(1), y: +r.y.toFixed(1), w: +r.width.toFixed(1), h: +r.height.toFixed(1) }; };
    out.roe = roes.map(rb); out.shiso = shisos.map(rb); out.preCount = pre.length; out.preHasRoe = pre.some((c) => c.includes("salmon-roe"));
    out.layerInsideDough = layer ? dough.contains(layer) : null;
    out.doughFilter = getComputedStyle(dough).filter; out.layerFilter = layer ? getComputedStyle(layer).filter : null;
    out.doughClass = dough.className;
    return out;
  });
  await page.addStyleTag({ content: `.pizza-dough{background:none!important;border-color:transparent!important;box-shadow:none!important}
    .pizza-dough-shape,.pizza-dough-square,.pizza-bake-overlay,.pizza-char-spots,.pizza-smoke-wrap,.pizza-perfect-glow,.pizza-sauce-heatmap{display:none!important}
    .pizza-topping:not(.pizza-post-layer .pizza-topping){display:none!important}
    *{animation:none!important;transition:none!important} html,body,main,.app-frame{background:#fff!important}`, });
  await page.waitForTimeout(250);
  const box = await page.locator(".pizza-dough").boundingBox();
  const buf = await page.screenshot({ clip: box! });
  writeFileSync(`out/m/${name}-post-isolated.png`, buf);
  return { info, buf };
}

test("V1 ON build: safety + real rounds + identification + ring + R8", async ({ page: firstPage, context }) => {
  let page = firstPage;
  test.setTimeout(600_000);
  const ext = await guardNetwork(context);
  const pre = await safetyPreflight(page);
  R.safety = { ...pre, firebaseEnvSet: false };
  await boot(page, baseSave());
  R.tabs = await page.locator(".making-step-tabs > *").allInnerTexts();

  // --- 3 in-window bake states with real FINISH, pixel-isolated post pieces
  const buffers: Record<string, Buffer> = {};
  const post: Record<string, any> = {};
  for (const [label, v] of [["low-edge-59", 59], ["just-68", 68], ["high-edge-77", 77]] as const) {
    if (label !== "low-edge-59") page = await freshRound(context, page);
    await prepareIkura(page);
    await bake(page, v);
    await expect(page.getByTestId("finish-dock")).toBeVisible();
    R[`finishHint-${label}`] = await page.getByTestId("finish-hint").innerText();
    R[`finishChips-${label}`] = await page.locator(".ingredient-tray .ingredient-chip").allInnerTexts();
    await placePost(page, 2, 1);
    await page.screenshot({ path: `out/m/${label}-finish.png` });
    const iso = await isolatedPost(page, label);
    buffers[label] = iso.buf; post[label] = iso.info;
    await page.reload(); // drops the style tag; R8 journal completes the round -- used for the notice check only in the dedicated test below
    await page.waitForSelector(".app-frame");
    // the reload settled the round; acknowledge to leave a clean save for the next label
    const ack = page.getByRole("button", { name: "確認した" });
    if (await ack.count()) { R[`reloadNotice-${label}`] = (await page.locator("[data-round-notice]").innerText()).replace(/\s+/g, " "); await ack.click(); }
  }
  R.post = post;
  R.postPixelSha256 = Object.fromEntries(Object.entries(buffers).map(([k, b]) => [k, sha(b)]));
  const raw = await Promise.all(Object.values(buffers).map((b) => sharp(b).raw().toBuffer()));
  R.postPixelsIdenticalAcrossStates = raw.every((r: Buffer) => Buffer.compare(r, raw[0]) === 0);
  const rgbMean = async (b: Buffer, r: any) => { const s = await sharp(b).extract({ left: Math.max(0, Math.round(r.x - (post["just-68"].dough?.x ?? 0))), top: 0, width: 1, height: 1 }).raw().toBuffer(); return s; };
  void rgbMean;

  log("3 states done");
  // --- failed bakes skip FINISH
  for (const [label, v] of [["raw-20", 20], ["burnt-95", 95]] as const) {
    page = await freshRound(context, page);
    await prepareIkura(page);
    await bake(page, v);
    R[`failed-${label}`] = { finishDock: await page.getByTestId("finish-dock").count(), result: (await resultText(page)).slice(0, 160) };
    await page.screenshot({ path: `out/m/failed-${label}.png` });
  }

  // --- identification (Free Cooking, bake 68)
  async function ident(label: string, mk: (p: Page) => Promise<void>, opts: { roePre?: number; shisoPre?: number } = {}) {
    page = await freshRound(context, page);
    log("ident " + label); await prepareIkura(page, opts); log("prepared"); await bake(page, 68); log("baked");
    await expect(page.getByTestId("finish-dock")).toBeVisible();
    await mk(page); log("placed");
    await page.getByTestId("finish-complete").click();
    const t = await resultText(page);
    await page.screenshot({ path: `out/m/ident-${label}.png` });
    R[`ident-${label}`] = t.slice(0, 260);
  }
  await ident("correct-roe2-shiso1", (p) => placePost(p, 2, 1));
  await ident("missing-shiso", (p) => placePost(p, 2, 0));
  await ident("one-roe", (p) => placePost(p, 1, 1));
  await ident("all-pre-bake", async () => {}, { roePre: 2, shisoPre: 1 });
  await ident("nothing-in-finish", async () => {});

  // --- ring: try to exceed 8 non-sauce pieces (5 pre + 3 post = 8), count post pieces after extra taps
  page = await freshRound(context, page);
  await prepareIkura(page); await bake(page, 68);
  await placePost(page, 2, 1);
  const count = () => page.locator(".pizza-post-layer .pizza-topping").count();
  R.ringPostAfter3 = await count();
  expect(await pickChip(page, /いくら/)).toBe(true);
  for (const [x, y] of [[28, 50], [72, 48], [50, 84], [50, 18]]) { await tapDoughPercent(page, x, y); }
  R.ringPostAfterExtraTaps = await count();
  const dists = await page.evaluate(() => {
    const els = [...document.querySelectorAll(".pizza-post-layer .pizza-topping, .pizza-dough .pizza-topping")].map((e) => { const r = e.getBoundingClientRect(); return { c: e.className.replace(/pizza-topping( pizza-topping--)?/, ""), x: r.x + r.width / 2, y: r.y + r.height / 2, w: r.width }; });
    let min = Infinity; for (let i = 0; i < els.length; i += 1) for (let j = i + 1; j < els.length; j += 1) min = Math.min(min, Math.hypot(els[i].x - els[j].x, els[i].y - els[j].y));
    return { n: els.length, minCenterDistPx: +min.toFixed(1), sizes: els.map((e) => e.w.toFixed(1)).join(",") };
  });
  R.ringAllPieces = dists;
  await page.screenshot({ path: "out/m/ring-after-extra-taps.png" });

  R.external = ext;
  writeFileSync("out/m/results.json", JSON.stringify(R, null, 2));
  console.log(JSON.stringify(R, null, 2));
  expect(ext.length).toBe(0);
});
