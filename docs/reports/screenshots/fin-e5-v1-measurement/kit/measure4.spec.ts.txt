import { writeFileSync, mkdirSync } from "node:fs";
import { test, expect, type Page } from "@playwright/test";
import { DISCOVERY_LADDER } from "../src/data/discoveryLadder";
import { RECIPES, countsTowardLadder } from "../src/data/recipes";
import { LATE_RECIPES } from "../src/data/lateCatalog";
import { baseSave, guardNetwork, pickChip, safetyPreflight, completeDoughStep, tapDoughPercent, enterBakePaused, landNeedleAndTakeOut, KEY } from "./helpers";
mkdirSync("out/m4", { recursive: true });
const R: Record<string, any> = {};
const FB = (v: number) => ({ start: v - 1, end: v + 1 });
const lateIds = new Set(LATE_RECIPES.map((r) => r.id as string));

test("C Research path (the real player route to ikura), ON build", async ({ page, context }) => {
  test.setTimeout(300_000);
  const ext = await guardNetwork(context);
  R.safety = await safetyPreflight(page);
  const ordinary = RECIPES.filter((r) => !lateIds.has(r.id) && countsTowardLadder(r.id)).map((r) => r.id);
  const materials = DISCOVERY_LADDER.steps.flatMap((s) => s.ingredientIds as readonly string[]);
  R.ladder = { steps: DISCOVERY_LADDER.steps.length, lateStep: DISCOVERY_LADDER.steps.at(-1), ordinaryCredited: ordinary.length };
  const save = baseSave({ dex: ordinary.map((recipeId) => ({ recipeId, discovered: true, bestScore: 70, bestStars: 3, timesMade: 1 })), pitzBalance: 999, unlockedForShopIngredientIds: materials, discoveryHintFacts: {} });
  await page.evaluate(([k, v]) => localStorage.setItem(k, v), [KEY, JSON.stringify(save)] as const);
  await page.goto("./"); await page.waitForSelector(".app-frame");
  await page.getByRole("button", { name: /ピザ図鑑/ }).click();
  await page.waitForSelector(".dex-overlay");
  const section = page.locator(".dex-overlay__research"); await section.scrollIntoViewIfNeeded();
  R.researchButtons = await section.getByRole("button").allInnerTexts();
  R.researchSection = (await section.innerText()).replace(/\s+/g, " ").slice(0, 400);
  await page.screenshot({ path: "out/m4/dex-research.png" });
  await section.getByRole("button", { name: /を研究する/ }).first().click();
  await page.waitForSelector(".pizza-stage");
  R.tabs = await page.locator(".making-step-tabs > *").allInnerTexts();
  await completeDoughStep(page);
  await page.getByRole("button", { name: /次へ/ }).click();
  await page.getByRole("button", { name: /次へ/ }).click();
  expect(await pickChip(page, /クリームチーズ/)).toBe(true);
  await tapDoughPercent(page, 38, 50); await tapDoughPercent(page, 62, 50);
  await page.getByRole("button", { name: /次へ/ }).click();
  expect(await pickChip(page, /サーモン/)).toBe(true);
  await tapDoughPercent(page, 38, 32); await tapDoughPercent(page, 62, 32); await tapDoughPercent(page, 50, 70);
  await enterBakePaused(page); await landNeedleAndTakeOut(page, FB(68));
  await expect(page.getByTestId("finish-dock")).toBeVisible();
  expect(await pickChip(page, /いくら/)).toBe(true); await tapDoughPercent(page, 36, 40); await tapDoughPercent(page, 64, 40);
  expect(await pickChip(page, /大葉/)).toBe(true); await tapDoughPercent(page, 50, 56);
  await page.screenshot({ path: "out/m4/research-finish.png" });
  await page.getByTestId("finish-complete").click();
  await page.waitForSelector(".result-panel");
  R.result = (await page.locator(".result-panel").innerText()).replace(/\s+/g, " ").slice(0, 500);
  await page.screenshot({ path: "out/m4/research-result.png" });
  R.external = ext;
  writeFileSync("out/m4/results.json", JSON.stringify(R, null, 2)); console.log(JSON.stringify(R, null, 2));
  expect(ext.length).toBe(0);
});
