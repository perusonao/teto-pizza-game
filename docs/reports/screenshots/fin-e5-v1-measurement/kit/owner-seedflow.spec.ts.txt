import { test, expect } from "@playwright/test";
import { completeDoughStep, tapDoughPercent, enterBakePaused, landNeedleAndTakeOut } from "../e2e/gestures";
import { pickChip } from "../v1spec/helpers";
test("owner flow on a PRODUCTION-mode build (no test hook): seed.html -> Dex research -> FINISH -> ikura", async ({ page, context }) => {
  test.setTimeout(120_000);
  const ext: string[] = [];
  await context.route("**/*", (r) => { const u = new URL(r.request().url()); if (u.hostname !== "127.0.0.1") { ext.push(u.href); return r.abort(); } return r.continue(); });
  await page.goto("seed.html");
  await page.locator("#seed").click(); await page.locator("#go").click();
  await page.waitForSelector(".app-frame");
  await page.getByRole("button", { name: /ピザ図鑑/ }).click(); await page.waitForSelector(".dex-overlay");
  const section = page.locator(".dex-overlay__research"); await section.scrollIntoViewIfNeeded();
  console.log("research entries:", await section.getByRole("button", { name: /を研究する/ }).count());
  await section.getByRole("button", { name: /を研究する/ }).first().click(); await page.waitForSelector(".pizza-stage");
  console.log("tabs", await page.locator(".making-step-tabs > *").allInnerTexts());
  await completeDoughStep(page);
  await page.getByRole("button", { name: /次へ/ }).click(); await page.getByRole("button", { name: /次へ/ }).click();
  expect(await pickChip(page, /クリームチーズ/)).toBe(true); await tapDoughPercent(page, 38, 50); await tapDoughPercent(page, 62, 50);
  await page.getByRole("button", { name: /次へ/ }).click();
  expect(await pickChip(page, /サーモン/)).toBe(true); await tapDoughPercent(page, 38, 32); await tapDoughPercent(page, 62, 32); await tapDoughPercent(page, 50, 70);
  await enterBakePaused(page); await landNeedleAndTakeOut(page, { start: 67, end: 69 });
  await expect(page.getByTestId("finish-dock")).toBeVisible();
  expect(await pickChip(page, /いくら/)).toBe(true); await tapDoughPercent(page, 36, 40); await tapDoughPercent(page, 64, 40);
  expect(await pickChip(page, /大葉/)).toBe(true); await tapDoughPercent(page, 50, 56);
  await page.getByTestId("finish-complete").click(); await page.waitForSelector(".result-panel");
  console.log("RESULT", (await page.locator(".result-panel").innerText()).replace(/\s+/g, " ").slice(0, 160));
  console.log("bench route in a built bundle:", await page.goto("./?dev=postlayer").then(() => page.locator("[data-post-layer-bench]").count()));
  expect(ext).toEqual([]);
});
