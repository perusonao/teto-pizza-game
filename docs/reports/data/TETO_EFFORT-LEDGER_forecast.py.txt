# Owner-prompt forecast for the full addition (ingredients 70 / recipes 84 / mechanics). Run: python3 -I <this file>
# All unit counts are measured from main 348b1e4 + staged goldens; per-unit prompt costs are ASSUMPTIONS (see the report).
import json
units = {
  "ingredient_trains": 11,                 # B21-B31 (ingredients + the recipes that complete with them: 41)
  "recipe_only_recipes": 43,               # recipes needing no new ingredient
  "new_mechanic_prs": 14,                  # M1 3, M5 2, M3 3, M2 2, M4 2, M6 2
  "gate_on_items": 5,                      # STEAM #726/#738, DRIZZLE #579, SEAR #727, FRY #749, PREP #696 (Production ON decisions)
}
case = {
  #            recipes/PR, train prompts, mech-PR prompts, gate-ON prompts, bulk-decision prompts, kickoff prompts
  "optimistic": dict(rpp=8, train=1, mech=3, gate=2, decisions=2,  kickoff=3),
  "standard":   dict(rpp=6, train=2, mech=5, gate=3, decisions=4,  kickoff=6),
  "pessimistic":dict(rpp=4, train=3, mech=8, gate=5, decisions=8,  kickoff=12),
}
def ceil(a,b): return -(-a//b)
out = {}
for k,c in case.items():
    rtrains = ceil(units["recipe_only_recipes"], c["rpp"])
    trains = units["ingredient_trains"] + rtrains
    line1 = units["ingredient_trains"] * c["train"]                       # ingredients (+ their recipes)
    line2 = rtrains * c["train"]                                          # recipe-only
    line3 = units["new_mechanic_prs"] * c["mech"] + units["gate_on_items"] * c["gate"]
    shared = c["decisions"] + c["kickoff"]
    total = line1 + line2 + line3 + shared
    # S-3 (conditional merge delegation): the merge half of each train/mech/gate prompt disappears (~35% of the train/mech prompts)
    saved = round(0.35 * (line1 + line2 + line3))
    out[k] = dict(recipe_only_trains=rtrains, trains=trains, line1_ingredients=line1, line2_recipes=line2,
                  line3_mechanics=line3, shared=shared, total=total, total_with_S3=total - saved)
# --- Claude tokens (measured medians per batch from 4 attributable batches B12/B13/B19/B20-A: output 438k, cacheRead 289M, cost 51 USD) ---
base = dict(output_k=438, cache_read_m=289, usd=51)
mult = {"optimistic": (0.4, 1.0, 0.5), "standard": (1.0, 1.5, 0.5), "pessimistic": (2.0, 3.0, 1.0)}  # train, mechanic PR, gate item
for k, c in case.items():
    mt, mm, mg = mult[k]
    n_tr = out[k]["trains"]
    units_eq = n_tr*mt + units["new_mechanic_prs"]*mm + units["gate_on_items"]*mg
    units_eq *= 1.10   # +10% S4 maintenance / measurement sessions
    out[k]["claude_output_M"] = round(units_eq*base["output_k"]/1000, 1)
    out[k]["claude_cache_read_B"] = round(units_eq*base["cache_read_m"]/1000, 1)
    out[k]["claude_usd"] = round(units_eq*base["usd"])
    # prompts if the CURRENT operation continues (GitHub Owner comments median 4 / PR, +1 session start): 5 per train-equivalent
    out[k]["total_current_operation"] = round(5*n_tr*{"optimistic":0.6,"standard":1.0,"pessimistic":1.6}[k] + out[k]["line3_mechanics"] + out[k]["shared"])
# --- Feature-unit regrouping of line 3 (Owner 2026-11-08). Units: U1 dough+pan (M1+M2), U2 prep/fry/mid-bake (M5+M4), U3 enclose/structure (M3+M6), U4 existing-Gate Production ON (5 gates).
# Owner prompts per unit = 1 decision table (d) + HV-and-merge rounds (R x h; the merge ask is folded into the HV report prompt, S-2) + 1 Production-ON bundle (o).
# Rounds = distinct iPhone UI surfaces: U1 2, U2 3, U3 3, U4 5. Gates, stop conditions, HV, Codex and CI are NOT reduced; only separate design asks, re-explanations and merge-only prompts are.
fu = {"optimistic": dict(d=1, h=1, o=1), "standard": dict(d=1, h=2, o=1), "pessimistic": dict(d=2, h=3, o=2)}
rounds = {"U1": 2, "U2": 3, "U3": 3, "U4": 5}
for k, c in fu.items():
    l3 = sum(c["d"] + r*c["h"] + c["o"] for r in rounds.values())
    out[k]["line3_feature_units"] = l3
    out[k]["total_feature_units"] = out[k]["line1_ingredients"] + out[k]["line2_recipes"] + l3 + out[k]["shared"]
    out[k]["total_feature_units_with_S3"] = out[k]["total_feature_units"] - round(0.35*(out[k]["line1_ingredients"] + out[k]["line2_recipes"] + l3))
print(json.dumps(out, indent=1, ensure_ascii=False))
