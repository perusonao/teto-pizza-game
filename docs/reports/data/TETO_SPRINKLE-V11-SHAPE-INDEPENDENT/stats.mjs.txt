import { flakePts, ribbonPts, area } from "./paths.mjs";
const CANDS = {
  F3: { layout: {}, render: {} },
  S1_ratio_mid: { layout: { wr0: 0.33, wrSpan: 0.30 }, render: {} },
  S2_irregular: { layout: { vLo: 0.55, vSpan: 0.9 }, render: { corner: 0.2 } },
  S3_taper: { layout: {}, render: { taper: 0.5, end: 0.15 } },
  S4_bevel: { layout: {}, render: { bevel: true } },
  S5_length: { layout: { fLen0: 8, fLenSpan: 9, rLen0: 9.5, rLenSpan: 6 }, render: {} },
};
const N = 2000;
const q = (a, p) => { const s = [...a].sort((x, y) => x - y); return s[Math.min(s.length - 1, Math.floor(p * s.length))]; };
const mean = (a) => a.reduce((x, y) => x + y, 0) / a.length;
function widthAt(poly, minx, maxx, t) { const x0 = minx + (maxx - minx) * (t - 0.06), x1 = minx + (maxx - minx) * (t + 0.06); let lo = 1e9, hi = -1e9; for (const p of poly) if (p.x >= x0 && p.x <= x1) { lo = Math.min(lo, p.y); hi = Math.max(hi, p.y); } return hi > lo ? hi - lo : 0; }
const out = {};
for (const [name, c] of Object.entries(CANDS)) {
  globalThis.__OPTS = c.layout;
  const mod = await import(`./layout.ts?${name}`);
  const flakeAsp = [], flakeRect = [], flakePar = [], ribAsp = [], ribPar = [], lens = [], inkPer = [], flakeLen = [], ribLen = [], taperAsym = [], flakeShare = [];
  for (let i = 1; i <= N; i++) {
    const { shavings } = mod.sprinkleLayout(`topping-${i}`);
    let ink = 0, flakeInk = 0;
    for (const s of shavings) {
      let P;
      if (s.kind === "ribbon") P = ribbonPts(s.len, s.wid, s.bend, s.wobble, s.profile, c.render);
      else P = flakePts(s.len, s.wid, s.verts, c.render);
      const poly = P.poly, xs = poly.map((p) => p.x), ys = poly.map((p) => p.y);
      const minx = Math.min(...xs), maxx = Math.max(...xs), miny = Math.min(...ys), maxy = Math.max(...ys);
      const A = area(poly); ink += A; if (s.kind === "flake") flakeInk += A;
      const L = maxx - minx, W = maxy - miny;
      if (s.kind === "fragment") { lens.push(L); continue; }
      lens.push(L);
      const w20 = widthAt(poly, minx, maxx, 0.2), w50 = widthAt(poly, minx, maxx, 0.5), w80 = widthAt(poly, minx, maxx, 0.8);
      const par = w50 > 0 ? Math.min(w20, w80) / w50 : 0;
      if (s.kind === "flake") { flakeAsp.push(W / L); flakeRect.push(A / (L * W)); flakePar.push(par); flakeLen.push(L); taperAsym.push(w50 > 0 ? Math.abs(w20 - w80) / w50 : 0); }
      else { ribAsp.push(W / L); ribPar.push(par); ribLen.push(L); }
    }
    inkPer.push(ink); flakeShare.push(flakeInk / ink);
  }
  out[name] = {
    flake_aspect_med: +q(flakeAsp, 0.5).toFixed(2), flake_aspect_p10_p90: `${q(flakeAsp, 0.1).toFixed(2)}-${q(flakeAsp, 0.9).toFixed(2)}`,
    flake_rectangularity_med: +q(flakeRect, 0.5).toFixed(2), flake_parallel_med: +q(flakePar, 0.5).toFixed(2), flake_taper_asym_med: +q(taperAsym, 0.5).toFixed(2),
    ribbon_aspect_med: +q(ribAsp, 0.5).toFixed(2), ribbon_parallel_med: +q(ribPar, 0.5).toFixed(2),
    flake_len_med: +q(flakeLen, 0.5).toFixed(1), flake_len_p90: +q(flakeLen, 0.9).toFixed(1), ribbon_len_med: +q(ribLen, 0.5).toFixed(1), ribbon_len_max: +Math.max(...ribLen).toFixed(1),
    piece_len_p90_all: +q(lens, 0.9).toFixed(1), ink_per_piece: +mean(inkPer).toFixed(0),
  };
}
const base = out.F3.ink_per_piece;
for (const k of Object.keys(out)) out[k].ink_vs_F3 = `${((out[k].ink_per_piece / base - 1) * 100).toFixed(0)}%`;
console.log(JSON.stringify(out, null, 1));
