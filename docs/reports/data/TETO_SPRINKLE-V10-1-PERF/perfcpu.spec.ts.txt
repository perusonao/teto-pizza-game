const SUM = ["RasterTask", "Layerize", "Commit", "UpdateLayerTree", "PrePaint", "Paint", "CompositeLayers"];
async function oneRun(browser: any, count: number, mode: "A" | "B" | "C") {
  const ctx = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: Number(process.env.DSF ?? 1) });
  const page = await ctx.newPage();
  await start(page, count);
  const n = await page.locator(".pizza-topping--parmigiano").count();
  await page.evaluate((m) => {
    const st = document.createElement("style");
    st.textContent = m === "A" ? ".ingredient-glyph--parmigiano-shavings{filter:none !important}" : m === "C" ? ".ingredient-glyph--parmigiano-shavings{filter:blur(6px) !important}" : "";
    document.head.appendChild(st);
  }, mode);
  const cdp = await ctx.newCDPSession(page);
  await cdp.send("Emulation.setCPUThrottlingRate", { rate: 4 });
  const tracePath = `${OUT}/trace-${count}-${mode}-${Date.now()}.json`;
  const bcdp = await browser.newBrowserCDPSession();
  const procs = async () => { const r: any = await bcdp.send("SystemInfo.getProcessInfo"); const o: Record<string, number> = {}; for (const p of r.processInfo) o[p.type + ":" + p.id] = p.cpuTime; return o; };
  const cpu0 = await procs();
  await browser.startTracing(page, { path: tracePath, categories: ["devtools.timeline", "disabled-by-default-devtools.timeline", "cc", "gpu"] });
  await page.getByRole("button", { name: /次へ/ }).click(); // BAKE starts: bake filter / melt scale animate every frame
  const frames = await page.evaluate(() => new Promise<number[]>((resolve) => {
    const d: number[] = []; let last = performance.now(); const t0 = last;
    function tick(now: number) { d.push(now - last); last = now; if (now - t0 < 8000) requestAnimationFrame(tick); else resolve(d); }
    requestAnimationFrame(tick);
  }));
  const cpu1 = await procs();
  await browser.stopTracing();
  const cpuBy: Record<string, number> = {}; for (const k of Object.keys(cpu1)) { const t = k.split(":")[0]; cpuBy[t] = (cpuBy[t] ?? 0) + (cpu1[k] - (cpu0[k] ?? 0)); }
  const tr = JSON.parse(fs.readFileSync(tracePath, "utf8"));
  const ev = Array.isArray(tr) ? tr : tr.traceEvents;
  const tot: Record<string, number> = Object.fromEntries(SUM.map((k) => [k, 0]));
  for (const e of ev) if (e.ph === "X" && SUM.includes(e.name) && e.dur) tot[e.name] += e.dur / 1000;
  fs.unlinkSync(tracePath);
  frames.sort((a, b) => a - b);
  await ctx.close();
  return { count, n, mode, cpu_gpu: +(cpuBy["GPU"] ?? 0).toFixed(3), cpu_renderer: +(cpuBy["renderer"] ?? 0).toFixed(3), cpu_browser: +(cpuBy["browser"] ?? 0).toFixed(3), cpu_total: +Object.values(cpuBy).reduce((a, b) => a + b, 0).toFixed(3), cpu_types: Object.keys(cpuBy).join(","), frames: frames.length, f_median: +frames[Math.floor(frames.length / 2)].toFixed(1), f_p95: +frames[Math.floor(frames.length * 0.95)].toFixed(1), mean: +(frames.reduce((a, b) => a + b, 0) / frames.length).toFixed(1), ...Object.fromEntries(Object.entries(tot).map(([k, v]) => [k, Math.round(v)])) };
}
test("perf A/A2/B at 12 pieces", async ({ browser }) => {
  const results: unknown[] = [];
  const plan: [number, number][] = [[12, 4]];
  for (const [count, reps] of plan) {
    const orders: ("A" | "B" | "C")[][] = [["A", "B", "C", "A"], ["B", "C", "A", "A"], ["C", "A", "A", "B"], ["A", "A", "B", "C"]];
    for (let r = 0; r < reps; r++) {
      let idx = 0;
      for (const m of orders[r]) {
        const res = await oneRun(browser, count, m);
        (res as any).label = m === "A" ? (idx === 0 || (orders[r][0] !== "A" && idx === 1) ? "A" : "A") : "B";
        (res as any).rep = r; results.push(res); idx++;
        fs.writeFileSync(`${OUT}/perf-results-cpu-dsf${process.env.DSF ?? 1}.json`, JSON.stringify(results, null, 1));
      }
    }
  }
});
