def parse_blocks(path):
    """Split a TS data file into top-level 4-space `id:` blocks (recipes / ingredients / orders)."""
    blocks, cur = {}, None
    for line in open(path, encoding="utf-8"):
        m = re.match(r'^    id: "([^"]+)"', line)
        if m:
            cur = m.group(1)
            blocks[cur] = []
        if cur is not None:
            blocks[cur].append(line)
    return blocks


def recipe_info(blocks, src):
    out = {}
    for rid, lines in blocks.items():
        txt = "".join(lines)
        ings = re.findall(r'ingredientId: "([^"]+)"', txt)
        if rid.startswith("order-") or not ings:
            continue
        nm = re.search(r'nameJa: "([^"]+)"', txt)
        out[rid] = {"name": nm.group(1) if nm else "", "ing": ings, "src": src, "ladderCredit": 'ladderCredit: false' not in txt}
    return out

def ingredient_info(blocks, src):
    out = {}
    for iid, lines in blocks.items():
        txt = "".join(lines)
        cat = re.search(r'category: "([^"]+)"', txt)
        if not cat or iid.startswith("order-"):
            continue
        nm = re.search(r'nameJa: "([^"]+)"', txt)
        pv = re.search(r'pieceVisual: "([^"]+)"', txt)
        out[iid] = {"name": nm.group(1) if nm else "", "category": cat.group(1), "src": src, "svg": pv.group(1) if pv else ""}
    return out


def ingredient_oneline(path, src):
    """One-line rows `{ id: "x", category: "y", nameJa: "z", ... }` (late-catalog ingredients are written this way, B21 onward)."""
    out = {}
    for line in open(path, encoding="utf-8"):
        m = re.match(r'^\s*\{ id: "([^"]+)", category: "([^"]+)", nameJa: "([^"]+)"', line)
        if m and not m.group(1).startswith("order-"):
            pv = re.search(r'pieceVisual: "([^"]+)"', line)
            out[m.group(1)] = {"name": m.group(3), "category": m.group(2), "src": src, "svg": pv.group(1) if pv else ""}
    return out

def dep_status(rid, entry):
    """Port of applyOwnerDecisions' status mapping onto the rejudge `open` entries (verified against decisions.json byStatus below)."""
    verdict, kind, did = entry.split(":", 2)
    if verdict == "RULE_DECIDED_WORK": return "WORK"
    if verdict == "WAITS_MECHANIC": return "WAITS_MECHANIC"
    if verdict == "WAITS_IN_FLIGHT": return "WAITS_IN_FLIGHT"
    if verdict == "AXIS_CONDITIONAL": return "AXIS_WITH_M1"
    if verdict == "STAGED_PAIR": return "STAGED_PAIR"
    if verdict == "EXCEPTION_NEEDED": return "EXCEPTION_REVIEW"
    # OWNER_DECISION
    if kind == "OWNER_RULE" and did == "R-1": return "SOURCE_RESEARCH"
    if kind == "OWNER_RULE" and did == "R-5": return "RESOLVED"
    if kind == "ID_SUFFIXED": return "WORK" if rid in d4_ids else "OWNER_DECISION"
    if kind == "OWNER_RULE" and did == "R-6":
        return "RESOLVED" if rid in d5 else "OWNER_DECISION"
    if kind == "UNRESOLVED_TOKEN" and did in D3_TOKENS: return "CANDIDATE_PENDING"
    if kind == "OWNER_RULE" and did == "R-7":
        toks = [x for x in rj[rid]["open"] if x.startswith("OWNER_DECISION:UNRESOLVED_TOKEN:")]
        if toks and all(t.split(":", 2)[2] in D3_TOKENS for t in toks) and next(s for s in S_R if s["id"] == rid)["unresolvedRelatedIds"]:
            return "CANDIDATE_PENDING"
    return "OWNER_DECISION"

def chunk(ids, n):
    """Balanced chunks of at most n (e.g. 10 items, cap 8 -> 5+5, not 8+2)."""
    if not ids: return []
    k = -(-len(ids) // n)
    size = -(-len(ids) // k)
    return [ids[i:i + size] for i in range(0, len(ids), size)]

        f.write(f"# {title}\n\n{intro}\n\n")
        f.write("| " + " | ".join(heads) + " |\n|" + "|".join(["---"] * len(heads)) + "|\n")
        for r in rows:
            f.write("| " + " | ".join(md_cell(r.get(c, "")) for c in cols) + " |\n")
