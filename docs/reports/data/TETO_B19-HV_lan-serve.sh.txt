#!/usr/bin/env bash
# B19 iPhone Safari HV: build the seed branch and serve it on the PC's private LAN address ONLY (no 0.0.0.0, no tunnel,
# no port forwarding). Run from the repo root on the seed branch. Usage: bash lan-serve.sh [PORT] [LAN_IP]
# Stop with Ctrl-C. Not for Production; the build is Preview-mode (own save key, PREVIEW badge).
set -euo pipefail
PORT="${1:-4173}"
IP="${2:-}"
if [ -z "$IP" ]; then  # macOS: en0/en1; Linux: first private IPv4
  IP="$( (ipconfig getifaddr en0 || ipconfig getifaddr en1 || hostname -I | tr ' ' '\n' | grep -E '^(10\.|192\.168\.|172\.(1[6-9]|2[0-9]|3[01])\.)' | head -1) 2>/dev/null | head -1 )"
fi
case "$IP" in 10.*|192.168.*|172.1[6-9].*|172.2[0-9].*|172.3[01].*) ;; *) echo "No private LAN IP found (got '$IP'). Pass it as 2nd arg."; exit 1;; esac
BRANCH="$(git rev-parse --abbrev-ref HEAD)"; SHA="$(git rev-parse --short HEAD)"
[ -z "$(git status --porcelain)" ] || { echo "Working tree is dirty; stop."; exit 1; }
echo "branch=$BRANCH seed-SHA=$SHA  #783 base=abb3da5a  (expected seed HEAD 03a6c512 or a later docs-only commit)"
[ -d node_modules ] || npm ci
rm -rf dist-lan
VITE_PREVIEW_MODE=1 VITE_PREVIEW_PR=783 VITE_PREVIEW_SHA="$SHA" npx vite build --base=/ --outDir dist-lan
echo; echo "Open on the iPhone (same Wi-Fi, PRIVATE Safari tab):"
echo "  http://$IP:$PORT/?hv=b19-pesto-burrata"
echo "  http://$IP:$PORT/?hv=b19-crudaiola"
echo "Badge must read: PREVIEW · PR#783 · $SHA"
exec npx vite preview --outDir dist-lan --host "$IP" --port "$PORT" --strictPort
