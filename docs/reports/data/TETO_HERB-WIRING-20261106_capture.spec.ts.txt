async function inject(page: Page) {
  if (CAND === "none") return;
  await page.addInitScript(([svgs, meta]) => {
    const NS = "http://www.w3.org/2000/svg";
    const put = (host: Element, id: string) => {
      if ((host as HTMLElement).dataset.cand === id) return;
      for (const n of [...host.childNodes]) if (n.nodeType === 3 && (n as Text).data.trim() === meta[id].emoji) (n as Text).data = "";
      const s = document.createElementNS(NS, "svg");
      s.setAttribute("viewBox", "0 0 64 64"); s.setAttribute("class", "ingredient-glyph ingredient-glyph--candidate"); s.setAttribute("aria-hidden", "true");
      s.setAttribute("style", "width:1.2em;height:1.2em;margin:-0.1em"); s.innerHTML = svgs[id];
      host.appendChild(s); (host as HTMLElement).dataset.cand = id;
    };
    const apply = () => {
      for (const chip of document.querySelectorAll(".ingredient-chip")) {
        const name = chip.querySelector(".ingredient-chip__name")?.textContent;
        const id = Object.keys(meta).find((k) => meta[k].name === name);
        const host = chip.querySelector(".ingredient-chip__emoji");
        if (id && host) put(host, id);
      }
      for (const id of Object.keys(meta)) for (const host of document.querySelectorAll(`.pizza-topping--${id} .ingredient-piece-visual__emoji`)) put(host, id);
    };
    new MutationObserver(apply).observe(document, { subtree: true, childList: true, characterData: true });
    document.addEventListener("DOMContentLoaded", apply);
  }, [svgs, meta] as const);
}
